cmake_minimum_required(VERSION 3.20)
project(ProsperoCompoundDeferredConsumer LANGUAGES C CXX)
enable_testing()
get_filename_component(COMPOUND_DEFERRED_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
function(add_deferred_consumer)
    include("${COMPOUND_DEFERRED_ROOT}/library.cmake")
    add_executable(compound_deferred_consumer "${COMPOUND_DEFERRED_ROOT}/tests/test_frontend.cpp")
    target_link_libraries(compound_deferred_consumer PRIVATE prospero-compound)
    target_compile_definitions(compound_deferred_consumer PRIVATE PROSPERO_ENABLE_COMPOUND=1)
    target_compile_features(compound_deferred_consumer PRIVATE cxx_std_17)
    add_test(NAME compound.frontend.deferred COMMAND compound_deferred_consumer)
endfunction()
cmake_language(DEFER CALL add_deferred_consumer)
